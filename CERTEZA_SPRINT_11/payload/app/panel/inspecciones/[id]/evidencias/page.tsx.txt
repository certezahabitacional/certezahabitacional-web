import Link from "next/link";
import { notFound } from "next/navigation";
import { prisma } from "@/lib/prisma";
import { eliminarEvidencia, registrarEvidencia } from "./actions";

export default async function EvidenciasPage({
  params,
}: {
  params: Promise<{ id: string }>;
}) {
  const { id } = await params;
  const inspeccion = await prisma.inspeccion.findUnique({
    where: { id },
    include: {
      cliente: true,
      inmueble: true,
      fotografias: {
        include: { hallazgo: true },
        orderBy: { creadaEn: "desc" },
      },
      hallazgos: { orderBy: { creadoEn: "desc" } },
    },
  });

  if (!inspeccion) notFound();

  return (
    <main className="min-h-screen bg-slate-950 px-6 py-8 text-white">
      <div className="mx-auto max-w-7xl">
        <Link href={`/panel/inspecciones/${id}`} className="text-sm font-black text-cyan-300">
          ← Expediente
        </Link>
        <h1 className="mt-4 text-3xl font-black">Evidencia fotográfica</h1>
        <p className="mt-2 text-slate-400">
          {inspeccion.folio} · {inspeccion.inmueble?.alias ?? inspeccion.tipoInmueble}
        </p>

        <div className="mt-8 grid gap-8 xl:grid-cols-[380px_1fr]">
          <section className="h-fit rounded-3xl border border-white/10 bg-slate-900 p-6">
            <h2 className="text-xl font-black">Registrar evidencia</h2>
            <p className="mt-2 text-sm leading-6 text-slate-400">
              Esta etapa registra imágenes mediante una URL pública. La carga directa a Supabase Storage se integrará en el siguiente módulo.
            </p>
            <form action={registrarEvidencia} className="mt-6 space-y-4">
              <input type="hidden" name="inspeccionId" value={id} />
              <Campo name="url" label="URL pública de la imagen *" type="url" required />
              <label className="block">
                <span className="mb-2 block text-sm font-bold">Relacionar con hallazgo</span>
                <select name="hallazgoId" className="w-full rounded-2xl border border-white/10 bg-slate-950 px-4 py-3">
                  <option value="">Evidencia general</option>
                  {inspeccion.hallazgos.map((hallazgo) => (
                    <option key={hallazgo.id} value={hallazgo.id}>
                      {hallazgo.area} · {hallazgo.titulo}
                    </option>
                  ))}
                </select>
              </label>
              <Campo name="tomadaEn" label="Fecha y hora de captura" type="datetime-local" />
              <label className="block">
                <span className="mb-2 block text-sm font-bold">Descripción</span>
                <textarea name="descripcion" rows={3} className="w-full rounded-2xl border border-white/10 bg-slate-950 px-4 py-3" />
              </label>
              <button className="w-full rounded-full bg-cyan-400 px-5 py-3 font-black text-slate-950">
                Guardar evidencia
              </button>
            </form>
          </section>

          <section>
            {inspeccion.fotografias.length === 0 ? (
              <div className="rounded-3xl border border-white/10 bg-slate-900 p-14 text-center text-slate-400">
                Aún no hay evidencias registradas.
              </div>
            ) : (
              <div className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
                {inspeccion.fotografias.map((foto) => (
                  <article key={foto.id} className="overflow-hidden rounded-3xl border border-white/10 bg-slate-900">
                    <div className="relative aspect-[4/3] bg-slate-800">
                      <img src={foto.url} alt={foto.descripcion ?? "Evidencia"} className="h-full w-full object-cover" />
                    </div>
                    <div className="p-4">
                      {foto.hallazgo && <p className="text-xs font-black text-cyan-300">{foto.hallazgo.area}</p>}
                      <p className="mt-1 text-sm text-slate-300">{foto.descripcion ?? "Sin descripción"}</p>
                      <form action={eliminarEvidencia} className="mt-4">
                        <input type="hidden" name="id" value={foto.id} />
                        <input type="hidden" name="inspeccionId" value={id} />
                        <button className="text-xs font-black text-rose-300">Eliminar</button>
                      </form>
                    </div>
                  </article>
                ))}
              </div>
            )}
          </section>
        </div>
      </div>
    </main>
  );
}

function Campo({ name, label, type = "text", required = false }: { name: string; label: string; type?: string; required?: boolean }) {
  return (
    <label className="block">
      <span className="mb-2 block text-sm font-bold">{label}</span>
      <input name={name} type={type} required={required} className="w-full rounded-2xl border border-white/10 bg-slate-950 px-4 py-3" />
    </label>
  );
}
